# INSTALLED LOCATION
# After setup, the live config is stored at:
# %LOCALAPPDATA%\OutpostHLLSeeder\setup_files\config.txt
#
# =============================================================================
# THE OUTPOST HLL AUTO-SEEDER CONFIGURATION
# =============================================================================
# This file is ordinary text. Right-click -> Edit to change it in Notepad.
#
# IMPORTANT: START_TIME_UTC is UTC, NOT the computer's local time.
# Using UTC means every seeder starts at the SAME MOMENT worldwide.
#
# Example:
#   START_TIME_UTC=12:30
# is:
#   13:30 in the UK while BST (UTC+1) is in effect
#   08:30 US Eastern while EDT (UTC-4) is in effect
#
# Daylight-saving changes alter the LOCAL clock time, but the actual trigger
# moment remains fixed at 12:30 UTC.
# =============================================================================

START_TIME_UTC=04:40

# Automatically replace START_TIME_UTC with the current time published by
# The Outpost public API.
# false = use the START_TIME_UTC above exactly as you set it; no seed-time web
#         request is made.
# true  = check the public API and save valid changes as the new start time.
# Default is true for new installations so The Outpost can publish the daily seed time automatically.
# Later updates must preserve the installed user's existing true/false choice.
AUTO_UPDATE_SEED_TIME=true
# Launch HLL only when player population is STRICTLY BELOW this number.
START_BELOW_PLAYERS=60
# Once population is STRICTLY ABOVE this number, begin random shutdown rolls.
STOP_ABOVE_PLAYERS=80
# Chance of closing HLL on each population check once above the threshold.
# 10 means exactly a 1-in-10 chance on each check.
# With CHECK_INTERVAL_SECONDS=60, each seeder independently rolls once per minute.
SHUTDOWN_CHANCE_ONE_IN=10
# How often CRCON is checked while HLL is running.
CHECK_INTERVAL_SECONDS=60

# Current proven delay before sending SPACE to the HLL splash screen.
SPLASH_WAIT_SECONDS=60


# Wake the PC this many minutes BEFORE START_TIME_UTC.
# The normal scheduler still performs the population check at START_TIME_UTC.
# This gives Windows/network connectivity time to recover after sleep.
WAKE_BEFORE_MINUTES=2

# After a SCHEDULED seeding session that this seeder launched and later closed,
# optionally put the PC back to sleep. Manual "Run seeder now" runs never sleep
# the PC automatically.
RETURN_TO_SLEEP_AFTER_SEEDING=false
# Delay after HLL has been successfully closed before requesting Windows sleep.
# This gives Steam/HLL/network activity time to settle.
RETURN_TO_SLEEP_DELAY_SECONDS=60


# HLL SETTINGS PROTECTION
# Protect the user's HLL GameUserSettings.ini from the known reset/default state.
#
# true  = before this seeder launches HLL, compare the live settings file with
#         the shipped known-default reference. Non-default settings are backed
#         up locally. If the live file matches the known-default reference and
#         a prior non-default backup exists, that backup is restored.
# false = do not read, back up or restore HLL GameUserSettings.ini.
PROTECT_HLL_SETTINGS=true
#
# Leave blank for the normal per-user HLL location:
# %LOCALAPPDATA%\HLL\Saved\Config\WindowsNoEditor\GameUserSettings.ini
#
# Users with an unusual location may enter an explicit FULL path here.
# Environment variables such as %LOCALAPPDATA% are supported.
HLL_SETTINGS_FILE=

# SAFETY NOTE:
# The seeder automatically cancels a run if Hell Let Loose is already running,
# or if another active Steam game is detected. Steam itself may remain open.
# This is a built-in protection rather than a configurable option.

# REMOTE SCHEDULE
# Controlled by AUTO_UPDATE_SEED_TIME above.
# When false, START_TIME_UTC is user-controlled and the seed-time API is not called.
# When true, the seeder checks:
# https://the-outpost.ngrok.app/api/get_seed_start_time
# A valid returned HH:MM UTC value replaces START_TIME_UTC in the installed config.
# If the web call fails or is invalid, the existing value is kept unchanged.

# MANUAL RUN NOTE:
# Menu option 1 ("Run seeder now") performs the population check immediately.
# If AUTO_UPDATE_SEED_TIME=true, it may also refresh the saved daily schedule.
